# Colors

Tokens semânticos de cor. Fonte da verdade: `docs/design-system/TOKENS.md`. Os swatches abaixo lêem
os CSS custom properties reais definidos em `src/styles/tokens.css` — não há valor hex duplicado
aqui, então esta página nunca dessincroniza dos tokens.

export function Swatch({ cssVar, label }) {
  return (
    <div style={{ display: "flex", alignItems: "center", gap: 12, padding: "6px 0" }}>
      <div
        style={{
          width: 40,
          height: 40,
          borderRadius: 8,
          border: "1px solid var(--border-default)",
          background: `var(${cssVar})`,
          flexShrink: 0,
        }}
      />
      <div>
        <div style={{ fontFamily: "var(--font-mono)", fontSize: "0.8125rem" }}>{cssVar}</div>
        {label ? (
          <div style={{ fontSize: "0.75rem", color: "var(--text-muted)" }}>{label}</div>
        ) : null}
      </div>
    </div>
  );
}

export function Group({ title, children }) {
  return (
    <div style={{ marginBottom: 24 }}>
      <h3 style={{ marginBottom: 8 }}>{title}</h3>
      {children}
    </div>
  );
}
